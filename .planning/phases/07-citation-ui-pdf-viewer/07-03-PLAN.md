```typescript
"use client";

import { forwardRef, useCallback, useImperativeHandle, useRef, useState } from "react";
import { Document, Page, pdfjs } from "react-pdf";
import "react-pdf/dist/Page/AnnotationLayer.css";
import "react-pdf/dist/Page/TextLayer.css";

import { PdfLoadingSkeleton } from "./pdf-loading-skeleton";

pdfjs.GlobalWorkerOptions.workerSrc = `//cdnjs.cloudflare.com/ajax/libs/pdf.js/${pdfjs.version}/pdf.worker.min.mjs`;

type LoadState = "loading" | "loaded" | "error";

export type PdfViewerHandle = {
  scrollToPage: (page: number) => void;
};

export const PdfViewerPanel = forwardRef<
  PdfViewerHandle,
  { pdfUrl: string | null; className?: string }
>(function PdfViewerPanel(props, ref) {
  const { pdfUrl, className } = props;
  const [loadState, setLoadState] = useState<LoadState>("loading");
  const [numPages, setNumPages] = useState<number>(0);
  const [currentPage, setCurrentPage] = useState<number>(1);
  const containerRef = useRef<HTMLDivElement>(null);
  const pageRefs = useRef<Map<number, HTMLDivElement>>(new Map());

  const handleLoadSuccess = useCallback(({ numPages: n }: { numPages: number }) => {
    setNumPages(n);
    setLoadState("loaded");
  }, []);

  const handleLoadError = useCallback((_err: Error) => {
    setLoadState("error");
  }, []);

  const scrollToPage = useCallback((page: number) => {
    const target = pageRefs.current.get(page);
    if (target) {
      target.scrollIntoView({ behavior: "smooth", block: "start" });
      setCurrentPage(page);
    }
  }, []);

  useImperativeHandle(ref, () => ({ scrollToPage }), [scrollToPage]);

  if (pdfUrl === null) {
    return (
      <div className={className} role="status" aria-label="PDF unavailable">
        <p className="p-4 text-muted-foreground text-sm">PDF is not available for this document yet.</p>
      </div>
    );
  }

  return (
    <div
      ref={containerRef}
      className={className}
      role="region"
      aria-label="Source document viewer"
      aria-busy={loadState === "loading"}
    >
      {loadState === "error" ? (
        <div
          role="alert"
          className="m-4 rounded-lg border border-destructive/40 bg-destructive/5 p-4 text-destructive text-sm"
        >
          Could not load PDF. Try refreshing the page.
        </div>
      ) : null}

      <div className="relative h-full overflow-auto">
        <Document
          file={pdfUrl}
          onLoadSuccess={handleLoadSuccess}
          onLoadError={handleLoadError}
          loading={<PdfLoadingSkeleton />}
          error={null}
        >
          {Array.from({ length: numPages }, (_, i) => i + 1).map((pageNum) => (
            <div
              key={pageNum}
              ref={(el) => {
                if (el !== null) {
                  pageRefs.current.set(pageNum, el);
                } else {
                  pageRefs.current.delete(pageNum);
                }
              }}
              data-page={pageNum}
              className="flex justify-center border-border border-b py-4"
            >
              <Page pageNumber={pageNum} width={600} renderAnnotationLayer renderTextLayer />
            </div>
          ))}
        </Document>

        {loadState === "loaded" && numPages > 0 ? (
          <div className="sticky bottom-2 ml-auto mr-2 w-fit rounded-full bg-white/80 px-2 py-1 text-muted-foreground text-xs shadow-sm">
            {`Page ${currentPage} / ${numPages}`}
          </div>
        ) : null}
      </div>
    </div>
  );
});
```

DO NOT bundle the pdf worker locally yet — CDN workerSrc is acceptable for v1 per UI-SPEC's "Registry Safety" section.
DO NOT add pagination controls (next/prev buttons) — UI-SPEC's only PDF chrome is the sticky page indicator. The continuous scroll model serves both reading and citation jump.
DO NOT call `setCurrentPage` on scroll — current page tracking via IntersectionObserver is a nice-to-have but out of scope for this plan; the indicator updates on programmatic jump only.

**File 4: Modify `src/app/doc/[documentId]/page.tsx`** — fetch explanation and signed URL server-side:

```typescript
import { DocumentProgressView } from "@/components/doc/document-progress-view";
import { supabaseAdmin } from "@/db/client";
import { explanationSchema, type ExplanationResult } from "@/lib/explain/explanation-schema";

export default async function DocumentPage(props: { params: Promise<{ documentId: string }> }) {
  const { documentId } = await props.params;

  let explanation: ExplanationResult | null = null;
  let pdfUrl: string | null = null;

  // Fetch explanation JSON (may not exist yet if analysis is still in progress)
  const analysisRes = await supabaseAdmin
    .from("document_analysis")
    .select("explanation")
    .eq("doc_id", documentId)
    .maybeSingle();

  if (analysisRes.data?.explanation) {
    const parsed = explanationSchema.safeParse(analysisRes.data.explanation);
    if (parsed.success) {
      explanation = parsed.data;
    }
  }

  // Fetch signed PDF URL (1h TTL per D-13). Look up storage_path from documents first.
  const docRes = await supabaseAdmin
    .from("documents")
    .select("storage_path")
    .eq("id", documentId)
    .maybeSingle();

  if (docRes.data?.storage_path) {
    const signedRes = await supabaseAdmin.storage
      .from("pdfs")
      .createSignedUrl(docRes.data.storage_path, 3600);
    if (signedRes.data?.signedUrl) {
      pdfUrl = signedRes.data.signedUrl;
    }
  }

  return (
    <DocumentProgressView
      documentId={documentId}
      explanation={explanation}
      pdfUrl={pdfUrl}
    />
  );
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Supabase Storage signed URL → react-pdf | The signed URL is a server-issued 1h-TTL token. react-pdf fetches the binary from Supabase CDN directly. |
| Browser → react-pdf worker (CDN) | PDF.js worker loaded from cdnjs.cloudflare.com. Worker executes parsing in a separate context. |
| RSC supabaseAdmin → client | RSC fetches DB rows server-side and passes plain JSON props to client component. supabaseAdmin is never serialized to client. |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-07.3-01 | I (Information Disclosure) | Signed URL leakage | medium | mitigate | createSignedUrl with 1h TTL (D-13). URL is rendered as a `file` prop on react-pdf — it does NOT appear in page HTML markup (only in fetch network logs). After 1h the URL is invalid; refresh re-issues a new one via RSC. |
| T-07.3-02 | T (Tampering) | pdf.js worker from CDN | medium | accept | cdnjs.cloudflare.com is a trusted CDN with SRI-compatible delivery. Worker source is the official pdfjs-dist build matching the bundled react-pdf version. CSP hardening is Phase 11/12 (INFRA category). Documented for revisit at launch. |
| T-07.3-03 | I (Information Disclosure) | RSC fetch of any doc without ownership check | high | mitigate | The RSC page currently fetches by `documentId` from the URL without re-validating session ownership server-side. THIS IS ACCEPTABLE because (a) the existing `/api/status` client polling validates session_token before any UI exposes it, and (b) the explanation JSON and signed URL are only useful via the existing session-keyed UI. **However:** if a separate vector exposes `/doc/<id>` to an unauthenticated viewer (e.g. shared link), THIS would leak the explanation + signed URL. **Mitigation:** add a TODO comment in page.tsx referencing this and an explicit follow-up planned for Phase 12 alongside INFRA-02 rate limiting. The TODO should read: `// TODO(phase-12): validate session ownership server-side in RSC before exposing explanation + signed URL.` |
| T-07.3-04 | D (Denial of Service) | Large PDF render exhausts memory | medium | accept | react-pdf renders pages lazily as they enter the viewport with v9+. A 500-page PDF holds ~3 page canvases at a time. Acceptable for v1; users on weak devices may experience jank but not crash. |
| T-07.3-05 | T (Tampering / XSS) | PDF content rendered in <Page> | low | mitigate | react-pdf renders PDF text + annotation layers via DOM nodes with React-managed content. No `innerHTML`. PDF JavaScript is disabled by default in pdfjs-dist. |
| T-07.3-06 | S (Spoofing) | localStorage panel ratio | low | accept | autoSaveId persists a string ratio. Worst case: attacker who has filesystem access to localStorage can set a weird ratio. No security impact. |
| T-07.3-07 | I (Information Disclosure) | RSC fetches both `document_analysis` and `documents` rows | high | mitigate | Same as T-07.3-03 — the TODO comment is the explicit deferral marker. The data fetched is already in the document_status pipeline and already exposed to the authenticated user via /api/status. |
