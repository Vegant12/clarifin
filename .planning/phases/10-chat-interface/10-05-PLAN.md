    ```typescript
    import { DocumentProgressView } from "@/components/doc/document-progress-view";
    import { supabaseAdmin } from "@/db/client";
    import { type ExplanationResult, explanationSchema } from "@/lib/explain/explanation-schema";
    import { type ScoreResult, scoreSchema } from "@/lib/explain/score-schema";
    import { fetchStockDataForDocument } from "@/lib/stock/fetch-stock-data";
    import type { ChartDataPoint, StockData } from "@/lib/stock/stock-schema";
    import { StarterQuestionsSchema } from "@/lib/chat/starter-questions-schema";
    import type { Message } from "ai";
    import { z } from "zod";

    const sessionIdSchema = z.string().uuid();

    export default async function DocumentPage(props: {
      params: Promise<{ documentId: string }>;
      searchParams: Promise<{ sessionId?: string }>;
    }) {
      const { documentId } = await props.params;
      const search = await props.searchParams;
      const sessionIdParsed = search.sessionId
        ? sessionIdSchema.safeParse(search.sessionId)
        : null;
      const sessionId: string | null =
        sessionIdParsed && sessionIdParsed.success ? sessionIdParsed.data : null;

      let explanation: ExplanationResult | null = null;
      let score: ScoreResult | null = null;
      let pdfUrl: string | null = null;
      let stockData: StockData | null = null;
      let chartData: ChartDataPoint[] | null = null;
      let stockError = false;
      let initialMessages: Message[] = [];
      let starterQuestions: string[] = [];

      // 1. Document analysis (now also reads starter_questions per Plan 01 migration)
      const analysisRes = await supabaseAdmin
        .from("document_analysis")
        .select("explanation, score_breakdown, starter_questions")
        .eq("doc_id", documentId)
        .maybeSingle();

      if (analysisRes.data?.explanation) {
        const parsed = explanationSchema.safeParse(analysisRes.data.explanation);
        if (parsed.success) explanation = parsed.data;
      }
      if (analysisRes.data?.score_breakdown) {
        const parsed = scoreSchema.safeParse(analysisRes.data.score_breakdown);
        if (parsed.success) score = parsed.data;
      }

      // 1b. CHAT-05 starter questions — prefer cached jsonb; trigger generation only if cache empty + explanation ready.
      const cached = analysisRes.data?.starter_questions;
      if (Array.isArray(cached)) {
        const v = StarterQuestionsSchema.safeParse({ questions: cached });
        if (v.success) starterQuestions = v.data.questions;
      }
      if (starterQuestions.length === 0 && explanation !== null) {
        try {
          const baseUrl =
            process.env.NEXT_PUBLIC_BASE_URL ?? "http://localhost:3000";
          const res = await fetch(`${baseUrl}/api/starter-questions`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ documentId }),
            cache: "no-store",
          });
          if (res.ok) {
            const body = (await res.json()) as { questions?: string[] };
            const v = StarterQuestionsSchema.safeParse({ questions: body.questions });
            if (v.success) starterQuestions = v.data.questions;
          }
        } catch {
          // Starter questions are non-essential; empty array degrades to "no pills" UI state.
        }
      }

      // 2. Document row (storage path + ticker — unchanged from prior phases)
      const docRes = await supabaseAdmin
        .from("documents")
        .select("storage_path, ticker")
        .eq("id", documentId)
        .maybeSingle();

      if (docRes.data?.storage_path) {
        const signedRes = await supabaseAdmin.storage
          .from("pdfs")
          .createSignedUrl(docRes.data.storage_path, 3600);
        if (signedRes.data?.signedUrl) pdfUrl = signedRes.data.signedUrl;
      }

      // 3. Stock data (unchanged — Phase 9)
      const ticker: string | null = docRes.data?.ticker ?? null;
      if (ticker !== null) {
        stockData = await fetchStockDataForDocument(documentId);
        if (stockData === null) stockError = true;
        else if (stockData.history.length > 0) chartData = stockData.history;
      }

      // 4. CHAT-04 7-day session restore — only when sessionId is in the URL.
      if (sessionId !== null) {
        const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString();
        const chatRes = await supabaseAdmin
          .from("chat_messages")
          .select("id, role, content, created_at")
          .eq("session_id", sessionId)
          .eq("doc_id", documentId)
          .gte("created_at", sevenDaysAgo)
          .order("created_at", { ascending: true })
          .limit(40);
        if (!chatRes.error && chatRes.data) {
          initialMessages = chatRes.data.map((r) => ({
            id: r.id,
            role: r.role as "user" | "assistant",
            content: r.content,
          }));
        }
      }

      return (
        <DocumentProgressView
          documentId={documentId}
          explanation={explanation}
          pdfUrl={pdfUrl}
          score={score}
          ticker={ticker}
          stockData={stockData}
          chartData={chartData}
          stockError={stockError}
          sessionId={sessionId}
          initialMessages={initialMessages}
          starterQuestions={starterQuestions}
        />
      );
    }
    ```

      })();
      return () => {
        cancelled = true;
      };
    }, [mounted, isSessionReady, sessionId, router, pathname]);

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| URL `?sessionId=` ↔ chat_messages query | A user-controlled URL parameter scopes a server-side database read |
| client localStorage session_token → /api/session → session_id → URL | Cross-storage identity flow |
| RSC fetch("/api/starter-questions") → loopback | Server-to-server HTTP call within the same Vercel deployment |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-23 | Spoofing | URL-supplied sessionId | accept | v1 sessions are anonymous UUIDs; any UUID is "valid" by design (matches existing pattern in /api/session). Worst case: a malicious URL reveals chat for a random session, but: (a) sessions are not secrets, (b) chat content is the user's own doc-scoped Q&A — no PII or auth claim |
| T-10-24 | Tampering | URL-supplied sessionId not a UUID | mitigate | sessionIdSchema = z.string().uuid().safeParse() in the RSC — invalid input falls back to `sessionId = null`, which short-circuits the chat_messages query AND renders the ChatPanel as a "Preparing your chat session…" placeholder until the client effect appends a valid sessionId |
| T-10-25 | Information Disclosure | cross-document chat leakage | mitigate | The Supabase chat_messages query uses BOTH `.eq("session_id", sessionId)` AND `.eq("doc_id", documentId)` — a user with session A viewing document B can never see chat from document C even if it shares session A (Pitfall 6 + T-10-12 from Plan 03) |
| T-10-26 | Denial of Service | unbounded chat history grows page bundle | mitigate | RSC query `.limit(40)` (= 20 turns) caps initialMessages payload; >7-day-old messages excluded via `.gte("created_at", sevenDaysAgo)` |
| T-10-27 | Tampering (prompt-injection via stored chat) | persisted chat content rendered in initialMessages | mitigate | Persisted content is the original user query OR the LLM's grounded reply — both already passed the guardrail. Plan 04 ChatMessage uses React text interpolation (no innerHTML) — stored content is never re-evaluated as instructions |
| T-10-28 | Information Disclosure | server-to-server fetch of /api/starter-questions exposes a public endpoint | accept | The endpoint reads document_analysis (already accessible via the RSC's direct supabaseAdmin call) and writes back to the same row; it does not return secrets. RSC could call generateObject directly, but the route exists for the client to refresh per-doc questions later (v2 use case) |
| T-10-29 | Tampering | DocumentProgressView client effect calls /api/session on every mount | accept | /api/session is idempotent (POST returns existing row if session_token already mapped) — no duplicate rows; latency is negligible (single index lookup) |
| T-10-30 | Repudiation | session_id → session_token mapping recovery | accept | Same v1 design as Phase 2 — recovery is via localStorage; if user clears localStorage, the prior chat is irretrievable but not corrupted. Documented as v1 limitation in CLAUDE.md "No Auth in v1" |
