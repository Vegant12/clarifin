       ```tsx
       import React, { Fragment } from "react";
       import { CitationInline } from "@/components/doc/citation-inline";
       import { parseCitations } from "@/lib/citations/parse-citations";

       export interface RenderInlineCitationsOptions {
         keyPrefix: string;
         docId: string;
         onGoToPage: (page: number) => void;
         /**
          * Optional transform applied to every TEXT leaf before it is emitted.
          * Used by ExplanationPanel to wrap glossary terms in JargonTooltip.
          * If omitted, text leaves are emitted as-is inside a Fragment.
          */
         transformText?: (text: string, key: string) => React.ReactNode;
       }

       export function renderInlineWithCitations(
         children: React.ReactNode,
         opts: RenderInlineCitationsOptions,
       ): React.ReactNode {
         const { keyPrefix, docId, onGoToPage, transformText } = opts;
         return React.Children.map(children, (child, childIdx) => {
           if (typeof child !== "string") return child;
           const tokens = parseCitations(child);
           if (tokens.length === 0) {
             // child is a non-empty string with no citations: still pass through transformText if provided
             if (transformText) {
               return (
                 <Fragment key={`${keyPrefix}.c${childIdx}`}>
                   {transformText(child, `${keyPrefix}.c${childIdx}`)}
                 </Fragment>
               );
             }
             return child;
           }
           return tokens.map((tok, tokIdx) => {
             const key = `${keyPrefix}.c${childIdx}.t${tokIdx}`;
             if (tok.kind === "citation") {
               return (
                 <CitationInline
                   key={key}
                   page={tok.page}
                   docId={docId}
                   onGoToPage={onGoToPage}
                 />
               );
             }
             if (transformText) {
               return <Fragment key={key}>{transformText(tok.value, key)}</Fragment>;
             }
             return <Fragment key={key}>{tok.value}</Fragment>;
           });
         });
       }
       ```

Re-read parseCitations behavior: for `"plain text"` it returns `[{kind:"text", value:"plain text"}]`, NOT `[]`. `[]` only happens for `""` or all-dropped malformed blocks. In the `[]` case for chat, returning the raw child string is correct (it might be `""` or `"[p.0]"`; chat will display it literally — acceptable since this is an edge case). For explanation, transformText running on `""` is harmless and on `"[p.0]"` would attempt jargon wrap on the literal string — also acceptable but rare.

       So the simpler rule: if `tokens.length === 0`, return `child` unchanged (no transformText).        if (tokens.length === 0) return child;
