```typescript
import { NextResponse } from "next/server";
import { z } from "zod";

import { supabaseAdmin } from "@/db/client";

const querySchema = z.object({
  doc_id: z.string().uuid(),
  session_token: z.string().uuid(),
  page: z.coerce.number().int().positive(),
});

export async function GET(request: Request): Promise<Response> {
  const url = new URL(request.url);
  const rawDocId = url.searchParams.get("doc_id");
  const rawSession = url.searchParams.get("session_token");
  const rawPage = url.searchParams.get("page");

  if (rawDocId === null || rawSession === null || rawPage === null) {
    return NextResponse.json(
      {
        error:
          "Missing query parameters. Required: doc_id (UUID), session_token (UUID), and page (positive integer).",
        missing: {
          doc_id: rawDocId === null,
          session_token: rawSession === null,
          page: rawPage === null,
        },
      },
      { status: 400 },
    );
  }

  const parsed = querySchema.safeParse({
    doc_id: rawDocId,
    session_token: rawSession,
    page: rawPage,
  });

  if (!parsed.success) {
    const docOk = z.string().uuid().safeParse(rawDocId);
    const sessOk = z.string().uuid().safeParse(rawSession);
    const pageOk = z.coerce.number().int().positive().safeParse(rawPage);
    const parts: string[] = [];
    if (!docOk.success) parts.push("doc_id must be a valid UUID");
    if (!sessOk.success) parts.push("session_token must be a valid UUID");
    if (!pageOk.success) parts.push("page must be a positive integer");
    return NextResponse.json(
      { error: parts.length > 0 ? parts.join("; ") : "Invalid query parameters." },
      { status: 400 },
    );
  }

  const { doc_id, session_token, page } = parsed.data;

  const sessionRes = await supabaseAdmin
    .from("chat_sessions")
    .select("id")
    .eq("session_token", session_token)
    .maybeSingle();

  if (sessionRes.error || !sessionRes.data) {
    return NextResponse.json({ error: "Document not found." }, { status: 404 });
  }

  const docRes = await supabaseAdmin
    .from("documents")
    .select("id")
    .eq("id", doc_id)
    .eq("session_id", sessionRes.data.id)
    .maybeSingle();

  if (docRes.error || !docRes.data) {
    return NextResponse.json({ error: "Document not found." }, { status: 404 });
  }

  const chunkRes = await supabaseAdmin
    .from("chunks")
    .select("content")
    .eq("doc_id", doc_id)
    .eq("page_number", page)
    .limit(1)
    .maybeSingle();

  if (chunkRes.error || !chunkRes.data) {
    return NextResponse.json({ error: "Page text not found." }, { status: 404 });
  }

  return NextResponse.json({ text: chunkRes.data.content });
}
```

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-07.1-01 | I (Information Disclosure) | /api/page-text | high | mitigate | Enforce session ownership: chat_sessions lookup by session_token → documents.session_id match before returning chunk content. Attacker who guesses a doc_id cannot read its page text without the matching session_token. Mirrors the proven /api/status pattern. |
| T-07.1-02 | T (Tampering) | /api/page-text query params | high | mitigate | zod schema strict-validates doc_id (UUID), session_token (UUID), and page (positive integer). Invalid input → 400 before any DB query. |
| T-07.1-03 | D (Denial of Service) | /api/page-text | medium | accept | Per-IP rate limiting is INFRA-02, planned for Phase 12. The route is read-only, indexed (doc_id + page_number), and returns one row max — DB cost per request is low. Re-evaluate at launch. |
| T-07.1-04 | I (Information Disclosure) | jargon-dictionary.json | low | accept | Static public data — financial term definitions are not sensitive. Bundled at build time, served as part of the client bundle. |
| T-07.1-05 | S (Spoofing) | session_token | medium | mitigate | session_token is a UUID generated server-side via POST /api/session (existing pattern). UUIDv4 entropy (122 bits) prevents enumeration. No new mitigation needed — inherits established posture. |
| T-07.1-06 | E (Elevation of Privilege) | supabaseAdmin import in route | high | mitigate | `import "server-only"` already enforced on `@/db/client` — bundler errors if any client file imports it. Route file is in `src/app/api/` so it is server-only by Next.js convention. |
